cmake_minimum_required(VERSION 3.16)
project(datasharing_examples LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "These DataSharing examples require Linux")
endif()
find_package(TRAVODDS 1 REQUIRED)
find_package(Threads REQUIRED)
add_library(datasharing_demo STATIC sample.cpp demo.cpp)
target_compile_features(datasharing_demo PUBLIC cxx_std_17)
target_include_directories(datasharing_demo PUBLIC ${TRAVODDS_INCLUDE_DIRS})
target_link_libraries(datasharing_demo PUBLIC TRAVODDS::TRAVODDS Threads::Threads)
foreach(name loan keyed mixed)
    add_executable(datasharing_${name} ${name}.cpp)
    target_link_libraries(datasharing_${name} PRIVATE datasharing_demo)
endforeach()
