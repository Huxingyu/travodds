cmake_minimum_required(VERSION 3.16)
project(travodds_lockfree_pool_prototype LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The prototype currently requires Linux futex/MAP_SHARED")
endif()
find_package(Threads REQUIRED)
add_executable(lockfree_pool_test test_pool.cpp)
add_executable(lockfree_pool_exec_demo exec_demo.cpp)
add_executable(lockfree_pool_controller_test test_controller.cpp)
foreach(target lockfree_pool_test lockfree_pool_exec_demo lockfree_pool_controller_test)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endforeach()
enable_testing()
add_test(NAME lockfree_pool COMMAND lockfree_pool_test)
add_test(NAME lockfree_pool_exec COMMAND lockfree_pool_exec_demo)
add_test(NAME lockfree_pool_controller COMMAND lockfree_pool_controller_test)
set_tests_properties(lockfree_pool lockfree_pool_exec lockfree_pool_controller PROPERTIES TIMEOUT 30)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME lockfree_pool_finite_model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/model_check.py)
    add_test(NAME lockfree_pool_registration_model COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/model_registration.py)
    set_tests_properties(lockfree_pool_finite_model PROPERTIES TIMEOUT 10)
    set_tests_properties(lockfree_pool_registration_model PROPERTIES TIMEOUT 10)
endif()
